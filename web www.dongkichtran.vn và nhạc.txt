    <script>
        const audio = document.getElementById('rapTrack');
        const lines = document.querySelectorAll('.line');

        // Bộ xử lý quét thời gian thực tế của bài nhạc để đổi màu chữ karaoke
        audio.addEventListener('timeupdate', () => {
            const currentTime = audio.currentTime;
            
            lines.forEach((line, index) => {
                const lineTime = parseFloat(line.getAttribute('data-time'));
                const nextLineTime = lines[index + 1] ? parseFloat(lines[index + 1].getAttribute('data-time')) : Infinity;

                if (currentTime >= lineTime && currentTime < nextLineTime) {
                    line.classList.add('highlight');
                    line.classList.remove('passed');
                } else if (currentTime >= nextLineTime) {
                    line.classList.remove('highlight');
                    line.classList.add('passed');
                } else {
                    line.classList.remove('highlight', 'passed');
                }
            });
        });
    </script>
